1//! Drawing non-layered tidy trees in linear time.
2//!
3//! A port of `src/non-layered-tidy-trees.c` in this repository, which is itself a
4//! translation of the algorithm described in
5//!
6//! > van der Ploeg, A. (2014), *Drawing non-layered tidy trees in linear time*,
7//! > Softw. Pract. Exper., 44, pages 1467–1484, doi:
8//! > [10.1002/spe.2213](https://doi.org/10.1002/spe.2213)
9//!
10//! The walks, the sibling chain, the threading and the one behavioural fix in
11//! [`separate`](fn@layout) follow the C sources line by line, so both produce
12//! bit-identical coordinates; what changed is the shape of the data, not the arithmetic:
13//!
14//! * nodes live in an [`Arena`] and refer to each other by [`NodeId`] rather than by
15//!   pointer, which is what makes the threads (`tl`/`tr`), the extreme nodes (`el`/`er`)
16//!   and the parent links expressible without `unsafe`;
17//! * the sibling chain is a `Vec` used as a stack instead of a hand-freed linked list;
18//! * the two callbacks are closures, so `walkud`/`cpairsud` are captured state rather
19//!   than `void *`, and they live in [`Callbacks`] instead of in the input struct — which
20//!   leaves [`LayoutInput`] plain data that [`layout`] takes by shared reference;
21//! * `free_tree` has no counterpart: dropping the [`Arena`] frees everything, without
22//!   recursion.
23//!
24//! ```
25//! use non_layered_tidy_trees::{layout, Arena, LayoutInput};
26//!
27//! let mut arena = Arena::new();
28//! let root = arena.add_node(1, 10.0, 10.0, 0.0, false);
29//! let a = arena.add_node(2, 10.0, 10.0, 0.0, false);
30//! let b = arena.add_node(3, 10.0, 10.0, 0.0, false);
31//! arena.set_children(root, &[a, b]);
32//!
33//! layout(&mut arena, &LayoutInput::new(root));
34//!
35//! assert_eq!((arena[a].x, arena[a].y), (0.0, 10.0));
36//! assert_eq!((arena[b].x, arena[b].y), (10.0, 10.0));
37//! assert_eq!((arena[root].x, arena[root].y), (5.0, 0.0)); // centred over its children
38//! ```

45/// A handle on a node held by an [`Arena`].

47/// Ids are only meaningful for the arena that produced them.

49pub struct NodeId(usize);

54        self.0
55    }

58/// A node of a hierarchy that acts as a tree, the counterpart of `tree_t`.
59///
60/// `w`, `h` and `margin` are the input; `x` and `y` are the output. Everything else is
61/// either bookkeeping filled in by [`layout`] (`level`, `childno`, `centeredxy`,
62/// `parent`) or private state of the algorithm.
63#[derive(Debug, Clone)]
64pub struct Node {
65    /// A caller-chosen label, one-based for the flat API (see [`layout_api`]).
66    pub idx: usize,
67    /// Distance from the root, set by [`layout`]; the root is at level 0.
68    pub level: usize,
69    /// Position among the parent's children, set by [`layout`]; 0 for the root.
70    pub childno: usize,
71    /// Whether `x`/`y` denote the middle of the box rather than its top left corner.
72    ///
73    /// Set by [`layout`] from [`LayoutInput::centeredxy`]; `false` until then.
74    pub centeredxy: bool,
75    /// Marks a node inserted only to take up space, as [`reify_flat_chunks`] does.
76    pub isdummy: bool,
77    /// Width of the box.
78    pub w: f64,
79    /// Height of the box.
80    pub h: f64,
81    /// Extra separation kept between this node and the sibling subtree to its right.
82    pub margin: f64,
83    /// Breadth or depth coordinate, depending on [`LayoutInput::vertically`].
84    pub x: f64,
85    /// Depth or breadth coordinate, depending on [`LayoutInput::vertically`].
86    pub y: f64,
87    /// The parent, set by [`layout`]; `None` for the root.
88    pub parent: Option<NodeId>,
89
90    children: Vec<NodeId>,
91
92    prelim: f64,
93    /// `mod` in the C sources, which is a keyword here.
94    modifier: f64,
95    shift: f64,
96    change: f64,
97    /// Left and right thread.
98    tl: Option<NodeId>,
99    tr: Option<NodeId>,
100    /// Extreme left and right nodes.
101    el: Option<NodeId>,
102    er: Option<NodeId>,
103    /// Sum of modifiers at the extreme nodes.
104    msel: f64,
105    mser: f64,
106}

120            y: 0.0,

122            children: Vec::new(),

124            modifier: 0.0,
125            shift: 0.0,
126            change: 0.0,
127            tl: None,
128            tr: None,
129            el: None,
130            er: None,
131            msel: 0.0,
132            mser: 0.0,

137    pub fn children(&self) -> &[NodeId] {

139    }

142/// The nodes of one or more trees, owning them for as long as the arena lives.
143///
144/// This replaces `init_tree`/`free_tree`: nodes are added with [`Arena::add_node`], wired
145/// together with [`Arena::set_children`], and freed all at once when the arena is dropped.
146#[derive(Debug, Clone, Default)]
147pub struct Arena {
148    nodes: Vec<Node>,
149}

151impl Index<NodeId> for Arena {

154    fn index(&self, id: NodeId) -> &Node {
155        &self.nodes[id.0]
156    }
157}

159impl IndexMut<NodeId> for Arena {
160    fn index_mut(&mut self, id: NodeId) -> &mut Node {

165impl Arena {
166    /// An empty arena.
167    pub fn new() -> Self {
168        Arena { nodes: Vec::new() }
169    }
170
171    /// An empty arena with room for `capacity` nodes.
172    pub fn with_capacity(capacity: usize) -> Self {
173        Arena {
174            nodes: Vec::with_capacity(capacity),
175        }
176    }
177
178    /// Adds a childless node, the counterpart of `init_tree`.
179    ///
180    /// `idx` is a label carried through to the output and is not interpreted, except by
181    /// the flat API where it has to be the one-based position of the node.
182    pub fn add_node(&mut self, idx: usize, w: f64, h: f64, margin: f64, isdummy: bool) -> NodeId {
183        self.nodes.push(Node::new(idx, w, h, margin, isdummy));
184        NodeId(self.nodes.len() - 1)
185    }
186
187    /// Appends one child to `parent`.
188    pub fn push_child(&mut self, parent: NodeId, child: NodeId) {
189        self.nodes[parent.0].children.push(child);
190    }
191
192    /// Replaces the children of `parent`, in left to right order.
193    pub fn set_children(&mut self, parent: NodeId, children: &[NodeId]) {
194        self.nodes[parent.0].children.clear();
195        self.nodes[parent.0].children.extend_from_slice(children);
196    }
197
198    /// The children of `t`, in left to right order.
199    pub fn children(&self, t: NodeId) -> &[NodeId] {
200        &self.nodes[t.0].children
201    }
202
203    /// How many nodes the arena holds.
204    pub fn len(&self) -> usize {
205        self.nodes.len()
206    }
207
208    /// Whether the arena holds no node at all.
209    pub fn is_empty(&self) -> bool {
210        self.nodes.is_empty()
211    }
212
213    /// The nodes, in creation order.
214    pub fn iter(&self) -> impl Iterator<Item = &Node> {
215        self.nodes.iter()
216    }
217
218    /// The ids of the subtree rooted at `t`, in preorder.
219    pub fn preorder(&self, t: NodeId) -> Vec<NodeId> {
220        let mut out = Vec::new();
221        self.preorder_into(t, &mut out);
222        out
223    }
224
225    fn preorder_into(&self, t: NodeId, out: &mut Vec<NodeId>) {
226        out.push(t);
227        for i in 0..self.nchildren(t) {
228            self.preorder_into(self.child(t, i), out);
229        }
230    }
231
232    /// The far edge of a node along the depth axis, the counterpart of `bottom`.
233    ///
234    /// That is the bottom of the box when laying out vertically and its right hand side
235    /// when laying out horizontally.
236    pub fn bottom(&self, t: NodeId, vertically: bool) -> f64 {
237        let n = &self.nodes[t.0];
238        if vertically {
239            n.y + if n.centeredxy { n.h / 2.0 } else { n.h }
240        } else {
241            n.x + if n.centeredxy { n.w / 2.0 } else { n.w }
242        }
243    }
244
245    fn child(&self, t: NodeId, i: usize) -> NodeId {
246        self.nodes[t.0].children[i]
247    }
248
249    fn nchildren(&self, t: NodeId) -> usize {
250        self.nodes[t.0].children.len()
251    }
252}

284/// Called once per node as the second walk finalizes it; the counterpart of `callback_t`.
285pub type WalkFn<'a> = dyn FnMut(&mut Arena, NodeId) + 'a;
286
287/// Called for every contour pair `separate` compares; the counterpart of `contourpairs_t`.
288pub type ContourPairsFn<'a> = dyn FnMut(&Arena, NodeId, NodeId, f64) + 'a;
289
290/// The two hooks of `treeinput_t`, as closures.
291///
292/// `walk` observes every node as `secondWalk` finalizes it — it is handed the arena, so it
293/// may move the node it is given, which does not perturb normalization. `contour_pairs`
294/// observes every pair of contour nodes `separate` compares, together with their `dist`;
295/// it is the only external window onto the contour walk.
296#[derive(Default)]
297pub struct Callbacks<'a> {
298    /// Called once per node, in preorder, during the second walk.
299    pub walk: Option<&'a mut WalkFn<'a>>,
300    /// Called for every contour pair `(sr, cl, dist)` compared while separating siblings.
301    pub contour_pairs: Option<&'a mut ContourPairsFn<'a>>,
302}

304/// A link of the chain of left siblings and their lowest depth coordinate.
305///
306/// The C keeps a hand-freed linked list; here the chain is a stack, so `nxt` is the
307/// element below and the head is the last one.
308#[derive(Debug, Clone, Copy)]
309struct ChainLink {
310    low: f64,
311    index: usize,
312}
313
314fn update_chain(chain: &mut Vec<ChainLink>, min: f64, i: usize) {
315    while let Some(top) = chain.last() {
316        if min >= top.low {
317            chain.pop();
318        } else {
319            break;
320        }
321    }
322    chain.push(ChainLink { low: min, index: i });
323}

325/// Process change and shift to add intermediate spacing to mod.
326fn add_child_spacing(a: &mut Arena, t: NodeId) {
327    let mut d = 0.0;
328    let mut modsumdelta = 0.0;
329    for i in 0..a.nchildren(t) {
330        let c = a.child(t, i);
331        d += a[c].shift;
332        modsumdelta += d + a[c].change;
333        a[c].modifier += modsumdelta;
334    }
335}
336
337fn move_subtree(a: &mut Arena, t: NodeId, i: usize, si: usize, dist: f64) {
338    let ci = a.child(t, i);
339    a[ci].modifier += dist;
340    a[ci].msel += dist;
341    a[ci].mser += dist;
342
343    if si + 1 != i {
344        // Are there intermediate children?
345        let nr = (i - si) as f64;
346        let ratio = dist / nr;
347        let cs = a.child(t, si + 1);
348        a[cs].shift += ratio;
349        a[ci].shift -= ratio;
350        a[ci].change -= dist - ratio;
351    }
352}

354fn next_left_contour(a: &Arena, t: NodeId) -> Option<NodeId> {
355    if a.nchildren(t) == 0 {
356        a[t].tl
357    } else {
358        Some(a.child(t, 0))
359    }
360}
361
362fn next_right_contour(a: &Arena, t: NodeId) -> Option<NodeId> {
363    let n = a.nchildren(t);
364    if n == 0 {
365        a[t].tr
366    } else {
367        Some(a.child(t, n - 1))
368    }
369}

371fn set_left_thread(a: &mut Arena, t: NodeId, i: usize, cl: NodeId, modsumcl: f64) {
372    let c0 = a.child(t, 0);
373    let ci = a.child(t, i);
374    let li = a[c0]
375        .el
376        .expect("the leftmost child has an extreme left node");
377    a[li].tl = Some(cl);
378
379    // Change mod so that the sum of modifier after following thread is correct.
380    let diff = (modsumcl - a[cl].modifier) - a[c0].msel;
381    a[li].modifier += diff;
382
383    // Change preliminary x coordinate so that the node does not move.
384    a[li].prelim -= diff;
385
386    // Update extreme node and its sum of modifiers.
387    a[c0].el = a[ci].el;
388    a[c0].msel = a[ci].msel;
389}
390
391/// Symmetrical to [`set_left_thread`].
392fn set_right_thread(a: &mut Arena, t: NodeId, i: usize, sr: NodeId, modsumsr: f64) {
393    let ci = a.child(t, i);
394    let cp = a.child(t, i - 1);
395    let ri = a[ci].er.expect("the i-th child has an extreme right node");
396    a[ri].tr = Some(sr);
397    let diff = (modsumsr - a[sr].modifier) - a[ci].mser;
398    a[ri].modifier += diff;
399    a[ri].prelim -= diff;
400    a[ci].er = a[cp].er;
401    a[ci].mser = a[cp].mser;
402}

404fn separate(
405    a: &mut Arena,
406    input: &LayoutInput,
407    cb: &mut Callbacks,
408    t: NodeId,
409    i: usize,
410    chain: &[ChainLink],
411) {
412    let mut sr = Some(a.child(t, i - 1));
413    let mut mssr = a[a.child(t, i - 1)].modifier;
414
415    let mut cl = Some(a.child(t, i));
416    let mut mscl = a[a.child(t, i)].modifier;
417
418    // The head of the chain is its last element; `ih->nxt` is the element below.
419    let mut ih = chain.len() - 1;
420
421    let mut first = true;
422
423    while let (Some(s), Some(c)) = (sr, cl) {
424        if a.bottom(s, input.vertically) > chain[ih].low {
425            ih = ih.checked_sub(1).expect(
426                "the sibling chain is seeded with a value that dominates the subtree depth",
427            );
428        }
429
430        // How far to the left of the right side of sr is the left side of cl?
431        let srd = if input.vertically { a[s].w } else { a[s].h };
432        let dist = (mssr + a[s].prelim + srd + a[s].margin) - (mscl + a[c].prelim);
433
434        // Pulling the subtree closer is only sound for the topmost pair of contour
435        // nodes, hence `first` has to fall on every iteration, moving or not.
436        if (first && dist < 0.0) || dist > 0.0 {
437            mscl += dist;
438            move_subtree(a, t, i, chain[ih].index, dist);
439        }
440
441        first = false;
442
443        if let Some(f) = cb.contour_pairs.as_mut() {
444            f(a, s, c, dist);
445        }
446
447        let sy = a.bottom(s, input.vertically);
448        let cy = a.bottom(c, input.vertically);
449
450        if sy <= cy {
451            sr = next_right_contour(a, s);
452            if let Some(n) = sr {
453                mssr += a[n].modifier;
454            }
455        }
456
457        if sy >= cy {
458            cl = next_left_contour(a, c);
459            if let Some(n) = cl {
460                mscl += a[n].modifier;
461            }
462        }
463    }
464
465    // Set threads and update extreme nodes.
466    // In the first case, the current subtree must be taller than the left siblings.
467    match (sr, cl) {
468        (None, Some(c)) => set_left_thread(a, t, i, c, mscl),
469        // In this case, the left siblings must be taller than the current tree.
470        (Some(s), None) => set_right_thread(a, t, i, s, mssr),
471        _ => {}
472    }
473}

475fn position_root(a: &mut Arena, t: NodeId, vertically: bool) {
476    // Position root between children, taking into account their mod.
477    let last = a.nchildren(t) - 1;
478    let first = a.child(t, 0);
479    let last = a.child(t, last);
480    let d = if vertically {
481        a[last].w - a[t].w
482    } else {
483        a[last].h - a[t].h
484    };
485    a[t].prelim =
486        (a[first].prelim + a[first].modifier + a[last].prelim + a[last].modifier + d) / 2.0;
487}

489fn first_walk(a: &mut Arena, input: &LayoutInput, cb: &mut Callbacks, t: NodeId) {
490    if a.nchildren(t) == 0 {
491        // setting extremes
492        a[t].el = Some(t);
493        a[t].er = Some(t);
494        a[t].msel = 0.0;
495        a[t].mser = 0.0;
496    } else {
497        let c0 = a.child(t, 0);
498        first_walk(a, input, cb, c0);
499
500        let mut chain = Vec::new();
501        let el = a[c0].el.expect("the first walk sets the extreme left node");
502        update_chain(&mut chain, a.bottom(el, input.vertically), 0);
503
504        for i in 1..a.nchildren(t) {
505            let ci = a.child(t, i);
506            first_walk(a, input, cb, ci);
507
508            let er = a[ci]
509                .er
510                .expect("the first walk sets the extreme right node");
511            let min = a.bottom(er, input.vertically);
512
513            separate(a, input, cb, t, i, &chain);
514
515            update_chain(&mut chain, min, i);
516        }
517
518        position_root(a, t, input.vertically);
519
520        // setting extremes
521        let last = a.child(t, a.nchildren(t) - 1);
522        a[t].el = a[c0].el;
523        a[t].msel = a[c0].msel;
524        a[t].er = a[last].er;
525        a[t].mser = a[last].mser;
526    }
527}

529fn second_walk(
530    a: &mut Arena,
531    input: &LayoutInput,
532    cb: &mut Callbacks,
533    t: NodeId,
534    modsum_init: f64,
535) -> f64 {
536    // keep it for the recursive call at the end.
537    let modsum = modsum_init + a[t].modifier;
538    let d = a[t].prelim + modsum;
539
540    let (xoffset, yoffset) = if input.centeredxy {
541        (a[t].w / 2.0, a[t].h / 2.0)
542    } else {
543        (0.0, 0.0)
544    };
545    a[t].centeredxy = input.centeredxy;
546
547    let mut best_min;
548
549    if input.vertically {
550        a[t].x = d + xoffset;
551        a[t].y += yoffset;
552        best_min = a[t].x;
553    } else {
554        a[t].x += xoffset;
555        a[t].y = d + yoffset;
556        best_min = a[t].y;
557    }
558
559    add_child_spacing(a, t);
560
561    if let Some(f) = cb.walk.as_mut() {
562        f(a, t);
563    }
564
565    for i in 0..a.nchildren(t) {
566        let c = a.child(t, i);
567        let current_min = second_walk(a, input, cb, c, modsum);
568        best_min = if current_min < best_min {
569            current_min
570        } else {
571            best_min
572        };
573    }
574
575    best_min
576}

578fn setup_walk(a: &mut Arena, input: &LayoutInput, t: NodeId, level: usize) {
579    a[t].level = level;
580    // initially the algorithm requires top-left wise coordinates.
581    a[t].centeredxy = false;
582
583    // Clear everything the three walks accumulate into, threads included, so that
584    // `layout` can be applied more than once to the same tree; the root starts the
585    // depth axis over at the origin, `input.x` and `input.y` place it elsewhere.
586    a[t].prelim = 0.0;
587    a[t].modifier = 0.0;
588    a[t].shift = 0.0;
589    a[t].change = 0.0;
590    a[t].msel = 0.0;
591    a[t].mser = 0.0;
592    a[t].tl = None;
593    a[t].tr = None;
594    a[t].el = None;
595    a[t].er = None;
596
597    if level == 0 {
598        if input.vertically {
599            a[t].y = 0.0;
600        } else {
601            a[t].x = 0.0;
602        }
603    }
604
605    let nextlevel = level + 1;
606
607    let b = a.bottom(t, input.vertically);
608
609    for i in 0..a.nchildren(t) {
610        let child = a.child(t, i);
611        a[child].childno = i;
612        a[child].parent = Some(t);
613
614        if input.vertically {
615            a[child].y = b;
616        } else {
617            a[child].x = b;
618        }
619
620        setup_walk(a, input, child, nextlevel);
621    }
622}
623
624fn third_walk(a: &mut Arena, t: NodeId, dx: f64, dy: f64) {
625    a[t].x -= dx;
626    a[t].y -= dy;
627
628    for i in 0..a.nchildren(t) {
629        let c = a.child(t, i);
630        third_walk(a, c, dx, dy);
631    }
632}

634/// Lays out the tree described by `input`, writing `x` and `y` into every node.
635///
636/// The drawing is normalized so that the smallest coordinate along the breadth axis ends
637/// up at `input.x` / `input.y`. Applying it twice to the same tree is well defined and
638/// gives the same coordinates, and `input` is never written to.
639pub fn layout(arena: &mut Arena, input: &LayoutInput) {
640    layout_with(arena, input, &mut Callbacks::default());
641}

643/// [`layout`], with the two observation hooks of `treeinput_t`.
644pub fn layout_with(arena: &mut Arena, input: &LayoutInput, cb: &mut Callbacks) {
645    let root = input.root;
646
647    setup_walk(arena, input, root, 0);
648    first_walk(arena, input, cb, root);
649    let minbreadth = second_walk(arena, input, cb, root, 0.0);
650
651    // `third_walk` subtracts, so subtracting the minimum breadth coordinate is the
652    // counterpart of the reference's `thirdWalk(t, -minX)`: it normalizes the drawing
653    // to start at the origin, which `input.x` and `input.y` then offset.
654    let dx = (if input.vertically { minbreadth } else { 0.0 }) - input.x;
655    let dy = (if input.vertically { 0.0 } else { minbreadth }) - input.y;
656
657    if dx != 0.0 || dy != 0.0 {
658        third_walk(arena, root, dx, dy);
659    }
660}

662/// Writes `[x, y, w, h]` per node into `array`, indexed by `idx - 1`.
663///
664/// `array` has to hold `4 * n` values, where `n` is the number of nodes of the subtree.
665pub fn flat_xywh_into(arena: &Arena, t: NodeId, array: &mut [f64]) {
666    for i in 0..arena.nchildren(t) {
667        flat_xywh_into(arena, arena.child(t, i), array);
668    }
669
670    let n = &arena[t];
671    let idx = (n.idx - 1) * 4;
672
673    array[idx] = n.x;
674    array[idx + 1] = n.y;
675    array[idx + 2] = n.w;
676    array[idx + 3] = n.h;
677}
678
679/// Writes the coordinates of `nodes` into `xy` as `[x_0.., y_0..]`, indexed by `idx - 1`.
680///
681/// `xy` has to hold `2 * nodes.len()` values.
682pub fn flat_xy_into(arena: &Arena, nodes: &[NodeId], xy: &mut [f64]) {
683    let n = nodes.len();
684
685    for &id in nodes {
686        let node = &arena[id];
687        xy[node.idx - 1] = node.x;
688        xy[node.idx - 1 + n] = node.y;
689    }
690}

692/// The deepest point of the subtree rooted at `t`, stopping at `to`.
693///
694/// `found` is set as soon as `to` is met among the descendants, which stops the walk.
695pub fn max_bottom(arena: &Arena, t: NodeId, to: NodeId, vertically: bool, found: &mut bool) -> f64 {
696    let mut m = arena.bottom(t, vertically);
697
698    let mut i = 0;
699    while !*found && i < arena.nchildren(t) {
700        let child = arena.child(t, i);
701        if child == to {
702            *found = true;
703            return m;
704        }
705        let b = max_bottom(arena, child, to, vertically, found);
706        m = if b > m { b } else { m };
707        i += 1;
708    }
709
710    m
711}
712
713/// The state threaded through [`max_bottom_between`], the counterpart of `fringemaxbottom_t`.
714#[derive(Debug, Clone, Copy)]
715pub struct FringeMaxBottom {
716    /// The deepest point met so far; the caller seeds it and reads it back.
717    pub bottom: f64,
718    /// Whether the depth axis is `y` rather than `x`.
719    pub vertically: bool,
720}
721
722/// The deepest point of everything that lies between `from` and `to` in the drawing.
723///
724/// Walks the siblings to the right of `from`, then climbs to the parent and repeats, until
725/// `to` is met.
726pub fn max_bottom_between(arena: &Arena, from: NodeId, to: NodeId, ud: &mut FringeMaxBottom) {
727    let p = match arena[from].parent {
728        Some(p) => p,
729        None => return,
730    };
731
732    let mut found = false;
733
734    let mut i = arena[from].childno + 1;
735    while !found && i < arena.nchildren(p) {
736        let child = arena.child(p, i);
737
738        if child == to {
739            found = true;
740            break;
741        }
742
743        let b = max_bottom(arena, child, to, ud.vertically, &mut found);
744
745        ud.bottom = if b > ud.bottom { b } else { ud.bottom };
746        i += 1;
747    }
748
749    if !found {
750        max_bottom_between(arena, p, to, ud);
751    }
752}

754/// Builds the arena of the flat description used by [`layout_api`].
755///
756/// Every node gets an invisible "gap" node inserted above it, which is what produces the
757/// spacing between levels: node `i` is `nodes[i]` and its gap node is `nodes[i + n]`, so
758/// the root of the resulting tree is `nodes[rooti - 1 + n]`.
759///
760/// # Panics
761///
762/// If the arrays are shorter than the layouts documented in [`layout_api`].
763pub fn reify_flat_chunks(
764    n: usize,
765    wh: &[f64],
766    whg: &[f64],
767    children: &[usize],
768) -> (Arena, Vec<NodeId>) {
769    assert!(
770        wh.len() >= 3 * n,
771        "`wh` holds w, h and margin for {n} nodes"
772    );
773    assert!(whg.len() >= 2 * n, "`whg` holds w and h for {n} gap nodes");
774    assert!(children.len() >= n, "`children` starts with {n} counts");
775
776    let mut arena = Arena::with_capacity(2 * n);
777    let mut nodes = Vec::with_capacity(2 * n);
778
779    for i in 0..n {
780        // the node with the content.
781        nodes.push(arena.add_node(i + 1, wh[i], wh[i + n], wh[i + 2 * n], false));
782    }
783
784    for i in 0..n {
785        // the node that separates.
786        let gap = arena.add_node(i + 1 + n, whg[i], whg[i + n], 0.0, true);
787        arena.push_child(gap, nodes[i]);
788        nodes.push(gap);
789    }
790
791    let mut nedges = n;
792
793    for i in 0..n {
794        for _ in 0..children[i] {
795            let c = children[nedges];
796            nedges += 1;
797            arena.push_child(nodes[i], nodes[c - 1 + n]);
798        }
799    }
800
801    (arena, nodes)
802}

804/// Lays out a tree given as parallel arrays, the counterpart of `layout_api`.
805///
806/// The array layouts, which the C leaves undocumented:
807///
808/// * `wh` — `3n` values, `[w_0..w_n-1, h_0..h_n-1, margin_0..margin_n-1]`; on return the
809///   first `2n` are overwritten with `[x_0.., y_0..]`
810/// * `whg` — `2n` values, the width and height of the invisible gap node inserted above
811///   every node, which is what produces the spacing between levels
812/// * `children` — `n` counts followed by the adjacency, one-based, grouped by parent
813/// * `rooti` — the one-based index of the root
814///
815/// # Panics
816///
817/// If the arrays are shorter than that, or if `rooti` is out of range.
818#[allow(clippy::too_many_arguments)]
819pub fn layout_api(
820    n: usize,
821    wh: &mut [f64],
822    whg: &[f64],
823    children: &[usize],
824    rooti: usize,
825    vertically: bool,
826    centeredxy: bool,
827    x: f64,
828    y: f64,
829) {
830    assert!(
831        (1..=n).contains(&rooti),
832        "`rooti` is a one-based index into {n} nodes"
833    );
834
835    let (mut arena, nodes) = reify_flat_chunks(n, wh, whg, children);
836
837    let root = nodes[rooti - 1 + n];
838
839    let input = LayoutInput {
840        root,
841        vertically,
842        centeredxy,
843        x,
844        y,
845    };
846
847    layout(&mut arena, &input);
848
849    flat_xy_into(&arena, &nodes[..n], wh);
850}
